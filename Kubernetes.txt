# STEP 1: Install Required Software
# Install Docker Desktop manually from:
# https://www.docker.com/products/docker-desktop/
# Start Docker Desktop before continuing.

winget install -e --id Kubernetes.kubectl
winget install -e --id Kubernetes.minikube

# Restart PowerShell after installation.

# STEP 2: Verify Installation
docker --version
kubectl version --client
minikube version

# STEP 3: Start Kubernetes Cluster
minikube start --driver=docker
kubectl get nodes

# STEP 4: Create Project Folder
mkdir kubernetes-demo
cd kubernetes-demo

# STEP 5: Create Simple Flask Application
@'
from flask import Flask

app = Flask(__name__)

@app.route("/")
def home():
    return "Hello from Kubernetes!"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
'@ | Set-Content app.py

# STEP 6: Create Requirements File
"flask" | Set-Content requirements.txt

# STEP 7: Create Dockerfile
@'
FROM python:3.9-slim
WORKDIR /app
COPY . /app
RUN pip install -r requirements.txt
EXPOSE 5000
CMD ["python", "app.py"]
'@ | Set-Content Dockerfile

# STEP 8: Build Docker Image
minikube image build -t flask-k8s-app .

# STEP 9: Create Kubernetes Deployment
@'
apiVersion: apps/v1
kind: Deployment
metadata:
  name: flask-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: flask-app
  template:
    metadata:
      labels:
        app: flask-app
    spec:
      containers:
      - name: flask-container
        image: flask-k8s-app
        imagePullPolicy: IfNotPresent
        ports:
        - containerPort: 5000
'@ | Set-Content deployment.yaml

# STEP 10: Create Kubernetes Service
@'
apiVersion: v1
kind: Service
metadata:
  name: flask-service
spec:
  type: NodePort
  selector:
    app: flask-app
  ports:
  - port: 5000
    targetPort: 5000
    nodePort: 30007
'@ | Set-Content service.yaml

# STEP 11: Deploy Application
kubectl apply -f deployment.yaml
kubectl apply -f service.yaml

# STEP 12: Check Cluster Status
kubectl get nodes
kubectl get pods
kubectl get deployments
kubectl get services

# STEP 13: Wait for Deployment
kubectl rollout status deployment/flask-deployment

# STEP 14: Open Application in Browser
minikube service flask-service

# EXPECTED OUTPUT:
# Hello from Kubernetes!

# STEP 15: View Application Logs
kubectl logs deployment/flask-deployment

# STEP 16: Delete Deployment and Service
# Run these commands after completing the demo:
# kubectl delete -f service.yaml
# kubectl delete -f deployment.yaml

# STEP 17: Stop Kubernetes Cluster
# minikube stop
